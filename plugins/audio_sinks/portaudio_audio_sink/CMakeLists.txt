cmake_minimum_required(VERSION 3.12)
project(portaudio_audio_sink)

if(MSVC)
    find_library(PORTAUDIO_LIBRARY portaudio_x64)
else()
    find_library(PORTAUDIO_LIBRARY portaudio)
endif()

if(PORTAUDIO_LIBRARY OR MSVC)
    message("Building with PortAudio sink")
    file(GLOB_RECURSE portaudio_audio_sink *.cpp)
    add_library(portaudio_audio_sink SHARED ${portaudio_audio_sink})

    if(MSVC)
        target_link_libraries(portaudio_audio_sink PUBLIC gobdump_core portaudio_x64.dll)
        install(TARGETS portaudio_audio_sink DESTINATION ${CMAKE_INSTALL_LIBDIR}/gobdump/plugins RUNTIME_DEPENDENCY_SET gobdump_deps)
    else()
        target_link_libraries(portaudio_audio_sink PUBLIC gobdump_core ${PORTAUDIO_LIBRARY})
        install(TARGETS portaudio_audio_sink DESTINATION ${CMAKE_INSTALL_LIBDIR}/gobdump/plugins)
    endif()
else()
    message("PortAudio Library could not be found! Not building.")
endif()
